    <script>
        Chart.defaults.font.family = "'Sarabun', sans-serif";
        Chart.defaults.font.size = 14;
        Chart.defaults.color = '#64748b';
        Chart.defaults.plugins.tooltip.backgroundColor = 'rgba(15, 23, 42, 0.9)';
        Chart.defaults.plugins.tooltip.padding = 12;
        Chart.defaults.plugins.tooltip.cornerRadius = 8;

        const formatNum = (num) => num.toLocaleString('th-TH', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
        const formatInt = (num) => num.toLocaleString('th-TH');

        let globalData = []; 
        let lineChartInst = null;
        let compareChartInst = null; 
        let currentProcessedData = null;

        function processData(deptFilter, empFilter) {
            let filteredData = globalData;
            if (deptFilter !== 'ALL') filteredData = filteredData.filter(d => d.department === deptFilter);
            if (empFilter !== 'ALL') filteredData = filteredData.filter(d => d.id === empFilter);

            let sumRev = 0, sumCost = 0, sumProfit = 0, sumProfit2025 = 0;
            let mRev = new Array(12).fill(0), mCost = new Array(12).fill(0), mProfit = new Array(12).fill(0);
            let summaryMap = {};

            filteredData.forEach(d => {
                sumRev += (d.avgRevenue || 0); 
                sumCost += (d.avgCost || 0); 
                sumProfit += (d.totalProfit || 0);
                sumProfit2025 += (d.profit2025 || 0);

                for(let m=0; m<12; m++) {
                    mRev[m] += (d.monthlyRevenue[m] || 0); 
                    mCost[m] += (d.monthlyCost[m] || 0); 
                    mProfit[m] += (d.monthlyProfit[m] || 0);
                }

                let key = deptFilter === 'ALL' ? d.department : d.name;
                if (!summaryMap[key]) summaryMap[key] = { 
                    name: key, revenue: 0, cost: 0, profit: 0, profit2025: 0, 
                    count: 0, status: d.status 
                };
                
                summaryMap[key].revenue += (d.avgRevenue || 0);
                summaryMap[key].cost += (d.avgCost || 0);
                summaryMap[key].profit += (d.totalProfit || 0);
                summaryMap[key].profit2025 += (d.profit2025 || 0);
                summaryMap[key].count += 1;
            });

            let lastDataIndex = 11;
            while(lastDataIndex >= 0 && mRev[lastDataIndex] === 0 && mCost[lastDataIndex] === 0 && mProfit[lastDataIndex] === 0) {
                mRev[lastDataIndex] = null;
                mCost[lastDataIndex] = null;
                mProfit[lastDataIndex] = null;
                lastDataIndex--;
            }

            let summaryArray = Object.values(summaryMap).sort((a,b) => b.profit - a.profit);
            return {
                empCount: filteredData.length, totalRevenue: sumRev, totalCost: sumCost, 
                totalProfit: sumProfit, totalProfit2025: sumProfit2025,
                monthlyRevenue: mRev, monthlyCost: mCost, monthlyProfit: mProfit, 
                summaryData: summaryArray,
                viewMode: empFilter !== 'ALL' ? 'PERSON' : (deptFilter !== 'ALL' ? 'DEPT' : 'ALL')
            };
        }

        function updateDashboard(processedData) {
            currentProcessedData = processedData;

            document.getElementById('kpi-total-emp').innerText = formatInt(processedData.empCount);
            document.getElementById('kpi-total-revenue').innerText = formatNum(processedData.totalRevenue);
            document.getElementById('kpi-total-cost').innerText = formatNum(processedData.totalCost);
            document.getElementById('kpi-total-profit').innerText = formatNum(processedData.totalProfit);
            
            // ประเมิน Status อัตโนมัติในภาพรวม
            let statusText = "ไม่มีข้อมูล";
            let statusIconBg = "var(--slate-100)";
            let statusIconColor = "var(--slate-700)";
            let statusHtmlIcon = `<svg width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5"><circle cx="12" cy="12" r="10"/><line x1="12" y1="8" x2="12" y2="12"/><line x1="12" y1="16" x2="12.01" y2="16"/></svg>`;

            if(processedData.viewMode === 'PERSON' && processedData.summaryData.length > 0) {
                let empStatus = processedData.summaryData[0].status;
                if(empStatus) statusText = empStatus;
                else statusText = processedData.totalProfit > 0 ? "ต่อยอดกำไรได้" : "ต้องปรับปรุง";
            } else {
                if(processedData.totalProfit > 0) statusText = "ภาพรวมมีกำไร";
                else if (processedData.totalProfit < 0) statusText = "ภาพรวมขาดทุน";
                else statusText = "เท่าทุน";
            }

            if(statusText.includes("ขาดทุน") || statusText.includes("ไม่") || statusText.includes("ปรับปรุง") || statusText.includes("เฝ้าระวัง")) {
                statusIconBg = "var(--danger-light)";
                statusIconColor = "var(--danger)";
                statusHtmlIcon = `<svg width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5"><circle cx="12" cy="12" r="10"/><line x1="15" y1="9" x2="9" y2="15"/><line x1="9" y1="9" x2="15" y2="15"/></svg>`;
            } else if(statusText.includes("กำไร") || statusText.includes("ต่อยอด") || statusText.includes("Pass")) {
                statusIconBg = "var(--success-light)";
                statusIconColor = "var(--success)";
                statusHtmlIcon = `<svg width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5"><path d="M22 11.08V12a10 10 0 1 1-5.93-9.14"/><polyline points="22 4 12 14.01 9 11.01"/></svg>`;
            }
            
            document.getElementById('kpi-status-text').innerText = statusText;
            document.getElementById('kpi-status-icon').style.background = statusIconBg;
            document.getElementById('kpi-status-icon').style.color = statusIconColor;
            document.getElementById('kpi-status-icon').innerHTML = statusHtmlIcon;

            let diff = processedData.totalProfit - processedData.totalProfit2025;
            let percent = processedData.totalProfit2025 !== 0 ? (diff / Math.abs(processedData.totalProfit2025)) * 100 : 0;
            let growthHtml = diff > 0 
                ? `<span style="color: var(--success);">▲ +${formatNum(percent)}%</span>` 
                : `<span style="color: var(--danger);">▼ ${formatNum(percent)}%</span>`;
            
            document.getElementById('kpi-growth-compare').innerHTML = `ปี 2568: ${formatNum(processedData.totalProfit2025)} <span style="margin-left:8px">${growthHtml}</span>`;

            // กราฟ Trend
            const ctx1 = document.getElementById('lineChart').getContext('2d');
            if(lineChartInst) lineChartInst.destroy();
            lineChartInst = new Chart(ctx1, {
                type: 'line',
                data: {
                    labels: ['ม.ค.', 'ก.พ.', 'มี.ค.', 'เม.ย.', 'พ.ค.', 'มิ.ย.', 'ก.ค.', 'ส.ค.', 'ก.ย.', 'ต.ค.', 'พ.ย.', 'ธ.ค.'],
                    datasets: [
                        { label: 'รายได้', data: processedData.monthlyRevenue, borderColor: '#10b981', tension: 0.4 },
                        { label: 'ต้นทุน', data: processedData.monthlyCost, borderColor: '#ef4444', borderDash: [5, 5], tension: 0.4 },
                        { label: 'กำไรสุทธิ', data: processedData.monthlyProfit, borderColor: '#4f46e5', borderWidth: 3, tension: 0.4, fill: true, backgroundColor: 'rgba(79, 70, 229, 0.1)' }
                    ]
                },
                options: { responsive: true, maintainAspectRatio: false }
            });

            const compareContainer = document.getElementById('compare-container');
            const tableContainer = document.getElementById('table-container');
            
            if (processedData.viewMode === 'PERSON') {
                compareContainer.style.display = 'none'; 
                tableContainer.style.display = 'none';
            } else {
                compareContainer.style.display = 'block'; 
                tableContainer.style.display = 'block';
                let isDeptView = processedData.viewMode === 'DEPT';
                
                // กราฟแท่งเปรียบเทียบ
                if(compareChartInst) compareChartInst.destroy();
                compareChartInst = new Chart(document.getElementById('compareChart').getContext('2d'), {
                    type: 'bar',
                    data: { 
                        labels: processedData.summaryData.map(d => d.name), 
                        datasets: [
                            { label: 'กำไร 68', data: processedData.summaryData.map(d => d.profit2025), backgroundColor: '#94a3b8' },
                            { label: 'กำไร 69', data: processedData.summaryData.map(d => d.profit), backgroundColor: '#4f46e5' }
                        ] 
                    },
                    options: { responsive: true, maintainAspectRatio: false }
                });

                // ตาราง
                const thead = document.getElementById('table-header');
                const tbody = document.getElementById('table-body');
                
                thead.innerHTML = `
                    <th class="text-center">ลำดับ</th>
                    <th>${isDeptView ? 'ชื่อพนักงาน' : 'แผนก'}</th>
                    <th class="text-right">รายได้ (เฉลี่ย)</th>
                    <th class="text-right">ต้นทุน (เฉลี่ย)</th>
                    <th class="text-right">กำไรปี 2568</th>
                    <th class="text-right">กำไรปี 2569</th>
                    <th class="text-center">Status</th>
                `;

                tbody.innerHTML = '';
                processedData.summaryData.forEach((item, idx) => {
                    let badgeClass = 'badge-success';
                    let displayStatus = item.status;
                    
                    if(!displayStatus) {
                        displayStatus = item.profit > 0 ? 'ต่อยอดกำไร' : 'เฝ้าระวัง';
                    }
                    
                    if (displayStatus.includes('ไม่ต่อยอด') || displayStatus.includes('ไม่') || displayStatus.includes('ขาดทุน') || displayStatus.includes('เฝ้าระวัง')) {
                        badgeClass = 'badge-danger';
                    }

                    tbody.innerHTML += `
                        <tr>
                            <td class="text-center" style="color: var(--slate-700);">${idx + 1}</td>
                            <td>${item.name}</td>
                            <td class="text-right" style="color: var(--success);">${formatNum(item.revenue)}</td>
                            <td class="text-right" style="color: var(--danger);">${formatNum(item.cost)}</td>
                            <td class="text-right" style="color: var(--slate-700);">${formatNum(item.profit2025)}</td>
                            <td class="text-right" style="color: var(--primary); font-weight: 700;">${formatNum(item.profit)}</td>
                            <td class="text-center">
                                <span class="badge ${badgeClass}">${displayStatus}</span>
                            </td>
                        </tr>
                    `;
                });
            }
        }

        function handleFilterChange() {
            const dept = document.getElementById('filter-dept').value;
            const empSelect = document.getElementById('filter-emp');
            empSelect.innerHTML = '<option value="ALL">พนักงานทุกคนในกลุ่มที่เลือก</option>';
            
            let empList = dept === 'ALL' ? globalData : globalData.filter(d => d.department === dept);
            empList.forEach(emp => { empSelect.innerHTML += `<option value="${emp.id}">${emp.name}</option>`; });

            updateDashboard(processData(dept, 'ALL'));
        }

        function handleEmpFilterChange() {
            const dept = document.getElementById('filter-dept').value;
            const emp = document.getElementById('filter-emp').value;
            updateDashboard(processData(dept, emp));
        }

        function resetFilters() {
            document.getElementById('filter-dept').value = 'ALL';
            handleFilterChange(); 
        }

        function showError(message) {
            const loader = document.getElementById('loading-overlay');
            loader.innerHTML = `

                    <svg width="48" height="48" fill="none" stroke="var(--danger)" stroke-width="2" viewBox="0 0 24 24" style="margin: 0 auto 16px;"><circle cx="12" cy="12" r="10"/><line x1="12" y1="8" x2="12" y2="12"/><line x1="12" y1="16" x2="12.01" y2="16"/></svg>
                    <h3 style="color: var(--danger); font-size: 20px; margin-bottom: 8px;">ไม่สามารถดึงข้อมูลได้</h3>
                    <p style="color: var(--slate-700); font-weight: 400; font-size: 15px;">${message}</p>
                </div>
            `;
        }

        function downloadCSV() {
            if (!currentProcessedData) return;
            let isDeptView = currentProcessedData.viewMode === 'DEPT';
            let headers = ['ลำดับ', isDeptView ? 'ชื่อพนักงาน' : 'แผนก', 'รายได้ (เฉลี่ย)', 'ต้นทุน (เฉลี่ย)', 'กำไรปี 2568', 'กำไรปี 2569', 'Status'];
            let csvContent = "data:text/csv;charset=utf-8,\uFEFF" + headers.join(",") + "\n";

            currentProcessedData.summaryData.forEach((item, idx) => {
                let displayStatus = item.status;
                if(!displayStatus) {
                    displayStatus = item.profit > 0 ? 'ต่อยอดกำไร' : 'เฝ้าระวัง';
                }
                let row = [idx + 1, `"${item.name}"`, item.revenue, item.cost, item.profit2025, item.profit, `"${displayStatus}"`];
                csvContent += row.join(",") + "\n";
            });

            const link = document.createElement('a');
            link.href = encodeURI(csvContent);
            link.download = `Financial_Analysis.csv`;
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }

        function initDashboard() {
            const loader = document.getElementById('loading-overlay');
            loader.style.display = 'flex';
            loader.style.opacity = '1';
            
            if (typeof google !== 'undefined' && google.script) {
                google.script.run.withSuccessHandler(function(result) {
                    if (result && result.status === 'success') {
                        globalData = result.rawData;
                        const deptSelect = document.getElementById('filter-dept');
                        deptSelect.innerHTML = '<option value="ALL">ภาพรวมบริษัท (ทุกแผนก)</option>';
                        result.departments.forEach(d => { deptSelect.innerHTML += `<option value="${d}">${d}</option>`; });
                        handleFilterChange(); 
                        
                        loader.style.opacity = '0';
                        setTimeout(() => loader.style.display = 'none', 500);
                    } else {
                        showError(result ? result.message : 'ไม่พบข้อมูลตอบกลับจากเซิร์ฟเวอร์');
                    }
                }).withFailureHandler(function(error) {
                    showError("ไม่สามารถเชื่อมต่อระบบหลังบ้านได้: " + (error.message || error));
                }).getSheetData();
            } else {
                showError("กรุณาเปิดระบบผ่านลิงก์ Web App ของ Google Apps Script เท่านั้น");
            }
        }

        window.onload = initDashboard;
    </script>
